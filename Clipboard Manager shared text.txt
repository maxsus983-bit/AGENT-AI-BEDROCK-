'use strict';

const fs = require('fs');
const path = require('path');

const context = require('../core/agent-context');
const logger = require('../core/logger');

class MemoryManager {

    constructor() {

        this.running = false;

        this.dataDirectory =
            path.join(
                process.cwd(),
                'data'
            );

        this.memoryFile =
            path.join(
                this.dataDirectory,
                'memory.json'
            );

        this.eventsFile =
            path.join(
                this.dataDirectory,
                'events.json'
            );

        this.playersFile =
            path.join(
                this.dataDirectory,
                'players.json'
            );

        this.commandsFile =
            path.join(
                this.dataDirectory,
                'commands.json'
            );

        this.chatFile =
            path.join(
                this.dataDirectory,
                'chat.json'
            );

        this.maxEvents =
            100000;

        this.maxChat =
            100000;

        this.maxCommands =
            50000;

        this.maxPlayers =
            10000;

        this.data = {

            facts: [],

            goals: [],

            preferences: [],

            important: [],

            observations: [],

            notes: []
        };

        this.events = [];

        this.players = {};

        this.commands = [];

        this.chat = [];

        this.saveTimer = null;
    }

    start() {

        if (this.running) {
            return;
        }

        this.running = true;

        this.ensureDirectory();

        this.load();

        this.startAutoSave();

        context.register(
            'memory',
            this
        );

        context.register(
            'memory-manager',
            this
        );

        logger.info(
            'Memory Manager ishga tushdi.'
        );
    }

    stop() {

        if (!this.running) {
            return;
        }

        this.save();

        this.stopAutoSave();

        this.running = false;

        logger.info(
            'Memory Manager to‘xtatildi.'
        );
    }

    ensureDirectory() {

        if (
            !fs.existsSync(
                this.dataDirectory
            )
        ) {

            fs.mkdirSync(
                this.dataDirectory,
                {
                    recursive: true
                }
            );
        }
    }

    load() {

        this.data =
            this.loadJSON(
                this.memoryFile,
                this.data
            );

        this.events =
            this.loadJSON(
                this.eventsFile,
                []
            );

        this.players =
            this.loadJSON(
                this.playersFile,
                {}
            );

        this.commands =
            this.loadJSON(
                this.commandsFile,
                []
            );

        this.chat =
            this.loadJSON(
                this.chatFile,
                []
            );

        logger.info(
            `Memory yuklandi: ${this.events.length} events, ${this.chat.length} chat`
        );
    }

    loadJSON(
        file,
        fallback
    ) {

        try {

            if (
                !fs.existsSync(
                    file
                )
            ) {

                return fallback;
            }

            const content =
                fs.readFileSync(
                    file,
                    'utf8'
                );

            if (
                !content.trim()
            ) {

                return fallback;
            }

            return JSON.parse(
                content
            );

        } catch (error) {

            logger.warn(
                `Memory faylini o‘qib bo‘lmadi: ${file}`
            );

            return fallback;
        }
    }

    save() {

        this.ensureDirectory();

        this.saveJSON(
            this.memoryFile,
            this.data
        );

        this.saveJSON(
            this.eventsFile,
            this.events
        );

        this.saveJSON(
            this.playersFile,
            this.players
        );

        this.saveJSON(
            this.commandsFile,
            this.commands
        );

        this.saveJSON(
            this.chatFile,
            this.chat
        );
    }

    saveJSON(
        file,
        data
    ) {

        const tempFile =
            `${file}.tmp`;

        try {

            fs.writeFileSync(
                tempFile,
                JSON.stringify(
                    data,
                    null,
                    2
                ),
                'utf8'
            );

            fs.renameSync(
                tempFile,
                file
            );

        } catch (error) {

            logger.error(
                `Memory save xatosi: ${error.message}`
            );
        }
    }

    startAutoSave() {

        this.stopAutoSave();

        this.saveTimer =
            setInterval(
                () => {

                    try {

                        this.save();

                    } catch (error) {

                        logger.error(
                            `Auto-save xatosi: ${error.message}`
                        );
                    }

                },
                30000
            );
    }

    stopAutoSave() {

        if (
            this.saveTimer
        ) {

            clearInterval(
                this.saveTimer
            );

            this.saveTimer =
                null;
        }
    }

    async remember(
        item
    ) {

        if (
            !item
        ) {

            return false;
        }

        const memory = {

            id:
                this.generateId(),

            timestamp:
                Date.now(),

            ...item
        };

        const type =
            String(
                item.type ||
                'notes'
            )
                .toLowerCase();

        if (
            !Array.isArray(
                this.data[type]
            )
        ) {

            this.data.notes.push(
                memory
            );

        } else {

            this.data[type].push(
                memory
            );
        }

        this.trimMainMemory();

        this.save();

        context.eventBus.emitSafe(
            'memory:remembered',
            memory
        );

        return memory;
    }

    addEvent(
        event
    ) {

        const item = {

            id:
                this.generateId(),

            timestamp:
                Date.now(),

            ...event
        };

        this.events.push(
            item
        );

        this.trimArray(
            this.events,
            this.maxEvents
        );

        context.eventBus.emitSafe(
            'memory:event',
            item
        );

        return item;
    }

    addChat(
        message
    ) {

        const item = {

            id:
                this.generateId(),

            timestamp:
                Date.now(),

            ...message
        };

        this.chat.push(
            item
        );

        this.trimArray(
            this.chat,
            this.maxChat
        );

        context.eventBus.emitSafe(
            'memory:chat',
            item
        );

        return item;
    }

    addCommand(
        command
    ) {

        const item = {

            id:
                this.generateId(),

            timestamp:
                Date.now(),

            ...command
        };

        this.commands.push(
            item
        );

        this.trimArray(
            this.commands,
            this.maxCommands
        );

        context.eventBus.emitSafe(
            'memory:command',
            item
        );

        return item;
    }

    updatePlayer(
        name,
        information = {}
    ) {

        if (
            !name
        ) {

            return null;
        }

        const key =
            String(
                name
            );

        const previous =
            this.players[key] ||
            {};

        this.players[key] = {

            ...previous,

            ...information,

            name:
                key,

            lastUpdated:
                Date.now(),

            seenCount:
                Number(
                    previous.seenCount ||
                    0
                ) + 1
        };

        const player =
            this.players[key];

        context.eventBus.emitSafe(
            'memory:player_updated',
            player
        );

        return player;
    }

    getPlayer(
        name
    ) {

        if (
            !name
        ) {

            return null;
        }

        return (
            this.players[
                String(name)
            ] ||
            null
        );
    }

    getPlayers() {

        return Object.values(
            this.players
        );
    }

    async search(
        query,
        limit = 30
    ) {

        const text =
            String(
                query ||
                ''
            )
                .toLowerCase()
                .trim();

        if (
            !text
        ) {

            return [];
        }

        const results = [];

        const collections = [

            ...this.data.facts,

            ...this.data.goals,

            ...this.data.preferences,

            ...this.data.important,

            ...this.data.observations,

            ...this.data.notes,

            ...this.events,

            ...this.chat,

            ...this.commands,

            ...Object.values(
                this.players
            )
        ];

        for (
            const item of collections
        ) {

            let serialized = '';

            try {

                serialized =
                    JSON.stringify(
                        item
                    )
                        .toLowerCase();

            } catch {
                continue;
            }

            if (
                serialized.includes(
                    text
                )
            ) {

                results.push(
                    item
                );
            }
        }

        results.sort(
            (
                a,
                b
            ) =>
                Number(
                    b.timestamp ||
                    b.lastUpdated ||
                    0
                ) -
                Number(
                    a.timestamp ||
                    a.lastUpdated ||
                    0
                )
        );

        return results.slice(
            0,
            Math.max(
                1,
                Number(limit) || 30
            )
        );
    }

    async recall(
        query
    ) {

        return this.search(
            query,
            50
        );
    }

    getRecentEvents(
        limit = 50
    ) {

        return this.events.slice(
            -Math.max(
                1,
                Number(limit) || 50
            )
        );
    }

    getRecentChat(
        limit = 50
    ) {

        return this.chat.slice(
            -Math.max(
                1,
                Number(limit) || 50
            )
        );
    }

    getRecentCommands(
        limit = 50
    ) {

        return this.commands.slice(
            -Math.max(
                1,
                Number(limit) || 50
            )
        );
    }

    getImportant() {

        return [
            ...this.data.important
        ];
    }

    getGoals() {

        return [
            ...this.data.goals
        ];
    }

    addGoal(
        goal
    ) {

        return this.remember({

            type:
                'goals',

            goal
        });
    }

    addFact(
        fact
    ) {

        return this.remember({

            type:
                'facts',

            fact
        });
    }

    addPreference(
        preference
    ) {

        return this.remember({

            type:
                'preferences',

            preference
        });
    }

    addImportant(
        information
    ) {

        return this.remember({

            type:
                'important',

            information
        });
    }

    forget(
        query
    ) {

        const text =
            String(
                query ||
                ''
            )
                .toLowerCase()
                .trim();

        if (
            !text
        ) {

            return {

                success:
                    false,

                removed:
                    0
            };
        }

        let removed = 0;

        const categories = [

            'facts',

            'goals',

            'preferences',

            'important',

            'observations',

            'notes'
        ];

        for (
            const category of
            categories
        ) {

            const old =
                this.data[category];

            const filtered =
                old.filter(
                    item => {

                        const textData =
                            JSON.stringify(
                                item
                            )
                                .toLowerCase();

                        if (
                            textData.includes(
                                text
                            )
                        ) {

                            removed++;

                            return false;
                        }

                        return true;
                    }
                );

            this.data[category] =
                filtered;
        }

        this.save();

        context.eventBus.emitSafe(
            'memory:forgotten',
            {
                query:
                    text,

                removed
            }
        );

        return {

            success:
                true,

            removed
        };
    }

    clearTemporary() {

        this.data.observations =
            [];

        this.events =
            [];

        this.save();

        return true;
    }

    statistics() {

        return {

            facts:
                this.data.facts.length,

            goals:
                this.data.goals.length,

            preferences:
                this.data.preferences.length,

            important:
                this.data.important.length,

            observations:
                this.data.observations.length,

            notes:
                this.data.notes.length,

            events:
                this.events.length,

            players:
                Object.keys(
                    this.players
                ).length,

            commands:
                this.commands.length,

            chat:
                this.chat.length
        };
    }

    trimMainMemory() {

        const limits = {

            facts:
                10000,

            goals:
                5000,

            preferences:
                5000,

            important:
                20000,

            observations:
                50000,

            notes:
                20000
        };

        for (
            const key of
            Object.keys(limits)
        ) {

            if (
                Array.isArray(
                    this.data[key]
                )
            ) {

                this.trimArray(
                    this.data[key],
                    limits[key]
                );
            }
        }
    }

    trimArray(
        array,
        max
    ) {

        if (
            array.length <= max
        ) {

            return;
        }

        array.splice(
            0,
            array.length - max
        );
    }

    generateId() {

        return (

            Date.now()
                .toString(36) +

            '-' +

            Math.random()
                .toString(36)
                .slice(2, 10)
        );
    }

    status() {

        return {

            running:
                this.running,

            storage:
                {

                    directory:
                        this.dataDirectory,

                    memory:
                        this.memoryFile,

                    events:
                        this.eventsFile,

                    players:
                        this.playersFile,

                    commands:
                        this.commandsFile,

                    chat:
                        this.chatFile
                },

            statistics:
                this.statistics()
        };
    }
}

const memory =
    new MemoryManager();

module.exports =
    memory;

module.exports.MemoryManager =
    MemoryManager;
