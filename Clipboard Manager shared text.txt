'use strict';

/*
===============================================================
 AKV EVENT BUS
===============================================================

Agent ichidagi barcha modullar shu tizim orqali gaplashadi.

Masalan:

Minecraft Client
      ↓
minecraft:chat
      ↓
World Observer
      ↓
observer:chat
      ↓
Memory
      ↓
AI Brain
      ↓
Decision
      ↓
Action Engine
      ↓
Minecraft Client

Shuningdek:

Telegram → Command Router → AI → Action Engine
GitHub   → Command Router → AI → Action Engine

Event Bus modullarni bir-biriga qattiq bog'lab qo'ymaslik
uchun ishlatiladi.

===============================================================
*/

const EventEmitter = require('events');

class AgentEventBus extends EventEmitter {

    constructor(options = {}) {

        super();

        /*
        Minecraft serverda eventlar juda ko'p bo'lishi mumkin.

        Default EventEmitter 10 listenerdan keyin warning beradi.
        Biz agent uchun ancha katta limit qo'yamiz.
        */

        this.setMaxListeners(
            Number(
                options.maxListeners
            ) || 200
        );

        this.historyLimit =
            Number(
                options.historyLimit
            ) || 1000;

        this.history = [];

        this.eventStatistics =
            new Map();

        this.startedAt =
            Date.now();
    }

    /* ========================================================
       SAFE EMIT
    ======================================================== */

    emitSafe(
        event,
        payload = {}
    ) {

        if (
            typeof event !== 'string' ||
            !event.trim()
        ) {

            return false;
        }

        const eventName =
            event.trim();

        const record =
            this.createRecord(
                eventName,
                payload
            );

        this.storeHistory(
            record
        );

        this.updateStatistics(
            eventName
        );

        try {

            return super.emit(
                eventName,
                payload
            );

        } catch (error) {

            /*
            Bitta event handler xatosi butun
            AI agentni yiqitmasligi kerak.
            */

            try {

                super.emit(
                    'system:handler_error',
                    {
                        event:
                            eventName,

                        payload,

                        error: {
                            name:
                                error?.name,

                            message:
                                error?.message,

                            stack:
                                error?.stack
                        },

                        timestamp:
                            Date.now()
                    }
                );

            } catch (_) {
                // Absolute safety.
            }

            return false;
        }
    }

    /* ========================================================
       ASYNC EMIT
    ======================================================== */

    async emitAsync(
        event,
        payload = {}
    ) {

        if (
            typeof event !== 'string' ||
            !event.trim()
        ) {

            return [];
        }

        const eventName =
            event.trim();

        const record =
            this.createRecord(
                eventName,
                payload
            );

        this.storeHistory(
            record
        );

        this.updateStatistics(
            eventName
        );

        const listeners =
            this.listeners(
                eventName
            );

        const results = [];

        for (
            const listener of listeners
        ) {

            try {

                const result =
                    await listener(
                        payload
                    );

                results.push({
                    success: true,
                    result
                });

            } catch (error) {

                results.push({
                    success: false,
                    error: {
                        name:
                            error?.name,

                        message:
                            error?.message,

                        stack:
                            error?.stack
                    }
                });

                /*
                Handler xatosi keyingi listenerlarga
                ta'sir qilmaydi.
                */

                try {

                    super.emit(
                        'system:handler_error',
                        {
                            event:
                                eventName,

                            payload,

                            error: {
                                name:
                                    error?.name,

                                message:
                                    error?.message,

                                stack:
                                    error?.stack
                            }
                        }
                    );

                } catch (_) {
                    // Ignore secondary error.
                }
            }
        }

        return results;
    }

    /* ========================================================
       CREATE EVENT RECORD
    ======================================================== */

    createRecord(
        event,
        payload
    ) {

        return {

            id:
                `${Date.now()}-${Math.random()
                    .toString(36)
                    .slice(2, 10)}`,

            event,

            timestamp:
                new Date()
                    .toISOString(),

            timestampMs:
                Date.now(),

            payload
        };
    }

    /* ========================================================
       HISTORY
    ======================================================== */

    storeHistory(record) {

        this.history.push(
            record
        );

        if (
            this.history.length >
            this.historyLimit
        ) {

            this.history.splice(
                0,
                this.history.length -
                this.historyLimit
            );
        }
    }

    getHistory(
        options = {}
    ) {

        const limit =
            Math.max(
                1,
                Number(
                    options.limit
                ) || 100
            );

        let result =
            this.history.slice(
                -limit
            );

        if (
            options.event
        ) {

            result =
                result.filter(
                    item =>
                        item.event ===
                        options.event
                );
        }

        if (
            options.prefix
        ) {

            result =
                result.filter(
                    item =>
                        item.event.startsWith(
                            options.prefix
                        )
                );
        }

        if (
            options.since
        ) {

            const since =
                Number(
                    options.since
                );

            result =
                result.filter(
                    item =>
                        item.timestampMs >=
                        since
                );
        }

        return result;
    }

    clearHistory() {

        this.history.length = 0;

        return true;
    }

    /* ========================================================
       STATISTICS
    ======================================================== */

    updateStatistics(
        event
    ) {

        const current =
            this.eventStatistics.get(
                event
            ) || 0;

        this.eventStatistics.set(
            event,
            current + 1
        );
    }

    getStatistics() {

        return Object.fromEntries(
            this.eventStatistics
        );
    }

    getEventCount(
        event
    ) {

        return (
            this.eventStatistics.get(
                event
            ) || 0
        );
    }

    /* ========================================================
       ONCE SAFE
    ======================================================== */

    onceSafe(
        event,
        listener
    ) {

        if (
            typeof listener !==
            'function'
        ) {

            throw new TypeError(
                'Listener function bo‘lishi kerak.'
            );
        }

        const wrapped =
            async payload => {

                try {

                    return await listener(
                        payload
                    );

                } catch (error) {

                    this.emitSafe(
                        'system:handler_error',
                        {
                            event,

                            error: {
                                name:
                                    error?.name,

                                message:
                                    error?.message,

                                stack:
                                    error?.stack
                            }
                        }
                    );

                    return undefined;
                }
            };

        return super.once(
            event,
            wrapped
        );
    }

    /* ========================================================
       ON SAFE
    ======================================================== */

    onSafe(
        event,
        listener
    ) {

        if (
            typeof listener !==
            'function'
        ) {

            throw new TypeError(
                'Listener function bo‘lishi kerak.'
            );
        }

        const wrapped =
            async payload => {

                try {

                    return await listener(
                        payload
                    );

                } catch (error) {

                    this.emitSafe(
                        'system:handler_error',
                        {
                            event,

                            error: {
                                name:
                                    error?.name,

                                message:
                                    error?.message,

                                stack:
                                    error?.stack
                            }
                        }
                    );

                    return undefined;
                }
            };

        super.on(
            event,
            wrapped
        );

        /*
        Keyinchalik listenerni olib tashlash uchun.
        */

        return () => {

            try {

                super.off(
                    event,
                    wrapped
                );

            } catch (_) {
                // Ignore.
            }
        };
    }

    /* ========================================================
       WAIT FOR EVENT
    ======================================================== */

    waitFor(
        event,
        options = {}
    ) {

        const timeout =
            Number(
                options.timeout
            ) || 30000;

        return new Promise(
            (resolve, reject) => {

                let timer = null;

                const cleanup = () => {

                    if (timer) {

                        clearTimeout(
                            timer
                        );

                        timer = null;
                    }

                    this.off(
                        event,
                        handler
                    );
                };

                const handler =
                    payload => {

                        cleanup();

                        resolve(
                            payload
                        );
                    };

                this.on(
                    event,
                    handler
                );

                timer =
                    setTimeout(
                        () => {

                            cleanup();

                            reject(
                                new Error(
                                    `Event timeout: ${event}`
                                )
                            );

                        },
                        timeout
                    );
            }
        );
    }

    /* ========================================================
       WAIT FOR ANY EVENT
    ======================================================== */

    waitForAny(
        events,
        timeout = 30000
    ) {

        if (
            !Array.isArray(events) ||
            events.length === 0
        ) {

            return Promise.reject(
                new Error(
                    'Events array bo‘sh.'
                )
            );
        }

        return new Promise(
            (resolve, reject) => {

                let timer = null;
                let finished = false;

                const handlers =
                    new Map();

                const cleanup = () => {

                    for (
                        const [
                            event,
                            handler
                        ]
                        of handlers.entries()
                    ) {

                        try {

                            this.off(
                                event,
                                handler
                            );

                        } catch (_) {
                            // Ignore.
                        }
                    }

                    handlers.clear();

                    if (timer) {

                        clearTimeout(
                            timer
                        );

                        timer = null;
                    }
                };

                const finish =
                    result => {

                        if (finished) {
                            return;
                        }

                        finished = true;

                        cleanup();

                        resolve(
                            result
                        );
                    };

                for (
                    const event of events
                ) {

                    const handler =
                        payload => {

                            finish({
                                event,
                                payload
                            });
                        };

                    handlers.set(
                        event,
                        handler
                    );

                    this.on(
                        event,
                        handler
                    );
                }

                timer =
                    setTimeout(
                        () => {

                            if (finished) {
                                return;
                            }

                            finished = true;

                            cleanup();

                            reject(
                                new Error(
                                    `Events timeout: ${events.join(', ')}`
                                )
                            );

                        },
                        timeout
                    );
            }
        );
    }

    /* ========================================================
       NAMESPACE
    ======================================================== */

    namespace(
        prefix
    ) {

        const base =
            String(prefix)
                .replace(/\.$/, '');

        return {

            emit:
                (
                    event,
                    payload = {}
                ) =>
                    this.emitSafe(
                        `${base}:${event}`,
                        payload
                    ),

            on:
                (
                    event,
                    listener
                ) =>
                    this.onSafe(
                        `${base}:${event}`,
                        listener
                    ),

            once:
                (
                    event,
                    listener
                ) =>
                    this.onceSafe(
                        `${base}:${event}`,
                        listener
                    ),

            waitFor:
                (
                    event,
                    options
                ) =>
                    this.waitFor(
                        `${base}:${event}`,
                        options
                    )
        };
    }

    /* ========================================================
       HEALTH
    ======================================================== */

    health() {

        return {

            running:
                true,

            startedAt:
                this.startedAt,

            uptime:
                Date.now() -
                this.startedAt,

            listenerCount:
                this.eventNames()
                    .reduce(
                        (
                            total,
                            event
                        ) =>
                            total +
                            this.listenerCount(
                                event
                            ),
                        0
                    ),

            registeredEvents:
                this.eventNames()
                    .map(
                        event =>
                            String(event)
                    ),

            historySize:
                this.history.length,

            statistics:
                this.getStatistics()
        };
    }

    /* ========================================================
       SHUTDOWN
    ======================================================== */

    shutdown() {

        this.removeAllListeners();

        this.clearHistory();

        this.eventStatistics.clear();

        return true;
    }
}

/* ============================================================
   DEFAULT EVENT BUS
=============================================================== */

const eventBus =
    new AgentEventBus({
        maxListeners: 200,
        historyLimit: 2000
    });

/* ============================================================
   EXPORT
=============================================================== */

module.exports =
    eventBus;

module.exports.AgentEventBus =
    AgentEventBus;
