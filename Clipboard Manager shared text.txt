'use strict';

const logger = require('../core/logger');

class CommandParser {

    constructor() {

        this.running = false;

        this.rules = [];

        this.registerDefaultRules();
    }

    start() {

        if (this.running) {
            return;
        }

        this.running = true;

        logger.info(
            'Command Parser ishga tushdi.'
        );
    }

    stop() {

        this.running = false;
    }

    parse(input, metadata = {}) {

        if (
            input === null ||
            input === undefined
        ) {

            return this.unknown('');
        }

        const original =
            String(input).trim();

        if (!original) {
            return this.unknown('');
        }

        const text =
            this.normalize(original);

        for (
            const rule of this.rules
        ) {

            const match =
                text.match(rule.pattern);

            if (!match) {
                continue;
            }

            try {

                const result =
                    rule.handler(
                        match,
                        original,
                        text,
                        metadata
                    );

                if (result) {

                    return {

                        success: true,

                        source:
                            original,

                        normalized:
                            text,

                        ...result
                    };
                }

            } catch (error) {

                logger.error(
                    `Command parser xatosi: ${error.message}`
                );
            }
        }

        return this.unknown(
            original
        );
    }

    register(
        name,
        pattern,
        handler,
        priority = 0
    ) {

        this.rules.push({

            name,
            pattern,
            handler,
            priority
        });

        this.rules.sort(
            (a, b) =>
                b.priority -
                a.priority
        );
    }

    extractPlayerName(
        text
    ) {

        const clean =
            String(text)
                .replace(
                    /kuzat|kuzatib tur|kuzatuvga ol/gi,
                    ''
                )
                .replace(
                    /\bni\b/gi,
                    ''
                )
                .trim();

        const match =
            clean.match(
                /([A-Za-z0-9_]+)/
            );

        return match
            ? match[1]
            : null;
    }

    normalize(
        text
    ) {

        return String(text)
            .toLowerCase()
            .trim()
            .replace(
                /[ʻʼ’‘]/g,
                "'"
            )
            .replace(
                /\s+/g,
                ' '
            );
    }

    unknown(
        input
    ) {

        return {

            success:
                false,

            recognized:
                false,

            action:
                'unknown',

            confidence:
                0,

            source:
                input,

            parameters: {},

            requiresAI:
                true
        };
    }

    explain(
        parsed
    ) {

        if (
            !parsed ||
            !parsed.success
        ) {

            return 'Buyruqni tushunish uchun AI tahlili kerak.';
        }

        const action =
            parsed.action;

        const params =
            parsed.parameters || {};

        switch (action) {

            case 'move':

                if (
                    params.target ===
                    'player'
                ) {

                    return `Player ${params.player} oldiga borish.`;
                }

                if (
                    params.target ===
                    'coordinates'
                ) {

                    return `(${params.x}, ${params.y}, ${params.z}) koordinatasiga borish.`;
                }

                return 'Harakat qilish.';

            case 'fight':

                return params.target
                    ? `${params.target} bilan jang qilish.`
                    : 'Jang qilish.';

            case 'watch':

                return `Player ${params.player || 'aniqlanmagan'}ni kuzatish.`;

            case 'watch_hidden':

                return `Player ${params.player || 'aniqlanmagan'}ni kuzatish rejimiga olish.`;

            case 'build':

                return 'Qurilish vazifasini bajarish.';

            case 'chat':

                return `Server chatiga: "${params.message}"`;

            case 'autonomous':

                return 'Mustaqil AI rejimini yoqish.';

            case 'report':

                return 'Dunyodagi joriy holat haqida hisobot tayyorlash.';

            case 'scan':

                return 'Atrofni tekshirish.';

            case 'remember':

                return 'Ma’lumotni xotiraga saqlash.';

            case 'recall':

                return 'Xotiradan ma’lumot qidirish.';

            default:

                return action;
        }
    }

    listRules() {

        return this.rules.map(
            rule => ({

                name:
                    rule.name,

                priority:
                    rule.priority,

                pattern:
                    String(
                        rule.pattern
                    )
            })
        );
    }

    status() {

        return {

            running:
                this.running,

            rules:
                this.rules.length
        };
    }
}
