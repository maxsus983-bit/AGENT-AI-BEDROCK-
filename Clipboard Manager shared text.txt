/* ============================================================

============================================================ */

class Logger {

    constructor() {

        this.logFile =
            path.join(
                ROOT,
                'data',
                'logs',
                'akv.log'
            );
    }

    timestamp() {
        return new Date().toISOString();
    }

    write(level, message, meta = null) {

        const line =
            `[${this.timestamp()}] [${level}] ${message}`;

        console.log(line);

        let output = line;

        if (meta !== null) {

            try {
                output += ` ${JSON.stringify(meta)}`;
            } catch (_) {}
        }

        try {

            fs.appendFileSync(
                this.logFile,
                output + '\n',
                'utf8'
            );

        } catch (error) {

            console.error(
                '[LOGGER] Log file error:',
                error.message
            );
        }
    }

    info(message, meta) {
        this.write('INFO', message, meta);
    }

    warn(message, meta) {
        this.write('WARN', message, meta);
    }

    error(message, meta) {
        this.write('ERROR', message, meta);
    }

    debug(message, meta) {

        if (
            CONFIG.app.environment !==
            'production'
        ) {
            this.write(
                'DEBUG',
                message,
                meta
            );
        }
    }

    success(message, meta) {
        this.write(
            'SUCCESS',
            message,
            meta
        );
    }
}

const logger = new Logger();

/* ============================================================
   EVENT BUS
============================================================ */

class AgentEventBus extends EventEmitter {

    constructor() {

        super();

        this.setMaxListeners(100);
    }

    emitSafe(event, payload = {}) {

        try {

            return this.emit(
                event,
                payload
            );

        } catch (error) {

            logger.error(
                `Event "${event}" handler failed`,
                {
                    error:
                        error.message
                }
            );

            return false;
        }
    }
}

    new AgentEventBus();

/* ============================================================
   SAFE JSON
============================================================ */

function safeJson(value) {

    try {

        return JSON.stringify(
            value,
            (_, current) => {

                if (
                    current instanceof Map
                ) {

                    return Object.fromEntries(
                        current
                    );
                }

                if (
                    current instanceof Set
                ) {

                    return Array.from(
                        current
                    );
                }

                if (
                    typeof current === 'bigint'
                ) {

                    return current.toString();
                }

                return current;
            },
            2
        );

    } catch (error) {

        return JSON.stringify({
            error:
                'serialization_failed',

            message:
                error.message
        });
    }
}

/* ============================================================
   EVENT PERSISTENCE
============================================================ */

function persistEvent(type, payload = {}) {

    const event = {

        id:
            `${Date.now()}-${Math.random()
                .toString(36)
                .slice(2)}`,

        timestamp:
            new Date().toISOString(),

        type,

        payload
    };

    state.recentEvents.push(event);

    if (
        state.recentEvents.length > 500
    ) {

        state.recentEvents.shift();
    }

    state.statistics.eventsSeen++;

    try {

        fs.appendFileSync(
            CONFIG.memory.events,
            safeJson(event) + '\n',
            'utf8'
        );

    } catch (error) {

        logger.error(
            'Event persistence failed',
            {
                error:
                    error.message
            }
        );
    }

    eventBus.emitSafe(
        'memory:event',
        event
    );

    return event;
}

function saveWorldSnapshot() {

    const snapshot = {

        timestamp:

            state.connected,

            state.spawned,

        position:
            state.position,

        rotation:
            state.rotation,

        health:

        hunger:
            state.hunger,

        currentTask:
            state.currentTask,

        currentGoal:
            state.currentGoal,

        autonomousMode:
            state.autonomousMode,

        observationMode:
            state.observationMode,

        hiddenObservation:
            state.hiddenObservation,

        players:
            Object.fromEntries(
                state.players
            ),

        entities:
            Object.fromEntries(
                state.entities
            ),

        combat:
            state.combat,

        statistics:
            state.statistics
    };

    try {

        fs.writeFileSync(
            CONFIG.memory.snapshot,
            safeJson(snapshot),
            'utf8'
        );

    } catch (error) {

        logger.error(
            'Could not save world snapshot',
            {
                error:
                    error.message
            }
        );
    }
}

/* ============================================================
   LOAD SNAPSHOT
============================================================ */

function loadWorldSnapshot() {

    if (
        !fs.existsSync(
            CONFIG.memory.snapshot
        )
    ) {

        return;
    }

    try {

        const raw =
            fs.readFileSync(
                CONFIG.memory.snapshot,
                'utf8'
            );

        const snapshot =
            JSON.parse(raw);

        if (snapshot.position) {
            state.position =
                snapshot.position;
        }

        if (snapshot.rotation) {
            state.rotation =
                snapshot.rotation;
        }

        if (snapshot.dimension) {
            state.dimension =
                snapshot.dimension;
        }

        if (snapshot.currentTask) {
            state.currentTask =
                snapshot.currentTask;
        }

        if (snapshot.currentGoal) {
            state.currentGoal =
                snapshot.currentGoal;
        }

        if (
            typeof snapshot.autonomousMode ===
            'boolean'
        ) {

            state.autonomousMode =
                snapshot.autonomousMode;
        }

        if (snapshot.statistics) {

            state.statistics = {
                ...state.statistics,
                ...snapshot.statistics
            };
        }

        logger.info(
            'Previous world snapshot restored.'
        );

    } catch (error) {

        logger.warn(
            'Could not restore previous snapshot',
            {
                error:
                    error.message
            }
        );
    }
}

    },

    register(name, module) {

        this.modules[name] =
            module;

        this.registry[name] =
            module;

        return module;
    }
};

/* ============================================================
   OPTIONAL MODULE LOADER
============================================================ */

function optionalRequire(modulePath) {

    try {

        if (
            !fs.existsSync(modulePath)
        ) {

            return null;
        }

        return require(modulePath);

    } catch (error) {

        logger.error(
            `Module loading failed: ${modulePath}`,
            {
                error:
                    error.message
            }
        );

        return null;
    }
}

/* ============================================================
   REGISTER MODULE
============================================================ */

function registerModule(name, module) {

    if (!module) {
        return;
    }

    agentContext.register(
        name,
        module
    );

    logger.debug(
        `Module registered: ${name}`
    );
}

/* ============================================================
   MODULE INITIALIZATION
============================================================ */

async function initializeModule(name, module) {

    if (!module) {

        logger.debug(
            `Module not installed yet: ${name}`
        );

        return null;
    }

    try {

        registerModule(
            name,
            module
        );

        if (
            typeof module.initialize ===
            'function'
        ) {

            await module.initialize(
                agentContext
            );

        } else if (
            typeof module.init ===
            'function'
        ) {

            await module.init(
                agentContext
            );
        }

        logger.success(
            `Module initialized: ${name}`
        );

        return module;

    } catch (error) {

        logger.error(
            `Module initialization failed: ${name}`,
            {
                error:
                    error.message,

                stack:
                    error.stack
            }
        );

        return null;
    }
}

        );

        logger.info(
            `Minecraft version: ${CONFIG.minecraft.version}`
        );

        logger.info(
            `Bot username: ${CONFIG.minecraft.username}`
        );

        logger.info(
            `Autonomous mode: ${state.autonomousMode}`
        );

        logger.info(

        logger.info(
            `Telegram enabled: ${CONFIG.telegram.enabled}`
        );

        /* ----------------------------------------------------
           RESTORE
        ---------------------------------------------------- */

           INITIALIZE MODULES
        ---------------------------------------------------- */

        const order = [

            'github'

        for (const name of order) {

            modules[name] =
                await initializeModule(
                    name,
                    modules[name]
                );
        }

        /* ----------------------------------------------------
           EVENTS
        ---------------------------------------------------- */

        eventBus.on(
            'minecraft:connected',
            data => {

                state.connected =
                    true;

                state.reconnecting =
                    false;

                logger.success(
                    'Minecraft connection established.'
                );

                persistEvent(
                    'minecraft_connected',
                    data
                );
            }
        );

        eventBus.on(
            'minecraft:disconnected',
            data => {

                state.connected =
                    false;

                state.spawned =
                    false;

                state.authenticated =
                    false;

                logger.warn(
                    'Minecraft connection lost.'
                );

                persistEvent(
                    'minecraft_disconnected',
                    data
                );

                saveWorldSnapshot();
            }
        );

        eventBus.on(
            'minecraft:spawned',
            data => {

                state.spawned =
                    true;

                logger.success(
                    'Bot spawned into Minecraft.'
                );

                persistEvent(
                    'minecraft_spawned',
                    data
                );
            }
        );

            'observer:player',
            player => {

                if (
                    !player ||
                    !player.name
                ) {
                    return;
                }

                state.players.set(
                    player.name,
                    player
                );

                state.statistics.playersSeen++;

                persistEvent(
                    'player_observation',
                    player
                );
            }
        );

        eventBus.on(
            'observer:chat',
            message => {

                state.recentChat.push(
                    message
                );

                if (
                    state.recentChat.length > 300
                ) {

                    state.recentChat.shift();
                }

                state.statistics.messagesSeen++;

                persistEvent(
                    'chat_message',
                    message
                );
            }
        );

        eventBus.on(
            'combat:attack',
            attack => {

                state.statistics.attacks++;

                persistEvent(
                    'combat_attack',
                    attack
                );
            }
        );

            'task:completed',
            task => {

                state.statistics.tasksCompleted++;

                    'task_completed',
                    task

                    'task_failed',
                    task
                );

        );

           SNAPSHOT TIMER

                    saveWorldSnapshot();

                        'Periodic snapshot failed',
                        {
                            error:
                                error.message
                        }
                    );
                }

        );

           STARTUP COMPLETE
        ---------------------------------------------------- */

            '===================================================='
        );

        logger.success(
            'AKV AI AGENT INITIALIZATION COMPLETE'

            '===================================================='

                    'function'
                ) {

                        agentContext
                    );

                } else if (
                    typeof modules.minecraft.start ===
                    'function'
                ) {

                    await modules.minecraft.start(
                        agentContext
                    );
                }

            } catch (error) {

                logger.error(
                    'Minecraft startup failed',
                    {
                        error:
                            error.message,

                        stack:
                            error.stack
                    }
                );
            }

        } else {

            logger.warn(
                'Minecraft client moduli topilmadi.'
            );
        }

        /* ----------------------------------------------------
           TELEGRAM
        ---------------------------------------------------- */

        if (
            CONFIG.telegram.enabled &&
            modules.telegram
        ) {

            try {

                if (
                    typeof modules.telegram.start ===
                    'function'
                ) {

                    );

                    );

                    'Telegram startup failed',

                        error:

            }

        /* ----------------------------------------------------
           GITHUB
        ---------------------------------------------------- */

                    typeof modules.github.start ===
                    'function'
                ) {

                    await modules.github.start(

                    );
                }

                    'GitHub startup failed',

                        error:

                    'agent:heartbeat',

                        timestamp:

                        combat:
                            state.combat
                    }
                );

            },
            1000
        );

    })();

    try {

        await startPromise;

    } finally {

        startPromise = null;
    }
}

/* ============================================================
   SHUTDOWN
============================================================ */

async function shutdown(signal) {

    if (state.shuttingDown) {
        return;
    }

    state.shuttingDown = true;

    logger.warn(
        `Shutdown requested: ${signal}`
    );

    try {

        saveWorldSnapshot();

        if (
            modules.commandAPI &&
            typeof modules.commandAPI.stop ===
            'function'
        ) {

            await modules.commandAPI.stop();
        }

        if (
            modules.telegram &&
            typeof modules.telegram.stop ===
            'function'
        ) {

            await modules.telegram.stop();
        }

        if (
            modules.github &&
            typeof modules.github.stop ===
            'function'
        ) {

            await modules.github.stop();
        }

        if (
            modules.minecraft &&
            typeof modules.minecraft.disconnect ===
            'function'
        ) {

            await modules.minecraft.disconnect();
        }

    } catch (error) {

        logger.error(
            'Shutdown error',
            {
                error:
                    error.message
            }
        );
    }

    logger.info(
        'AKV AI Agent stopped.'
    );

    process.exit(0);
}

/* ============================================================
   PROCESS EVENTS
============================================================ */

process.once(
    'SIGINT',
    () => shutdown('SIGINT')
);

process.once(
    'SIGTERM',
    () => shutdown('SIGTERM')
);

process.on(
    'uncaughtException',
    error => {

        logger.error(
            'UNCAUGHT EXCEPTION',
            {
                error:
                    error.message,

                stack:
                    error.stack
            }
        );
    }
);

process.on(
    'unhandledRejection',
    reason => {

        logger.error(
            'UNHANDLED PROMISE REJECTION',
            {
                reason:
                    reason instanceof Error
                        ? reason.message
                        : String(reason)
            }
        );
    }
);

/* ============================================================
   START
============================================================ */

start().catch(
    error => {

        logger.error(
            'FATAL STARTUP ERROR',
            {
                error:
                    error.message,

                stack:
                    error.stack
            }
        );

        process.exit(1);
    }
);
