'use strict';

const bedrock = require('bedrock-protocol');

const config = require('../config');
const context = require('../core/agent-context');
const logger = require('../core/logger');

class MinecraftClient {

    constructor() {

        this.client = null;

        this.connected = false;

        this.connecting = false;

        this.stopped = false;

        this.reconnectTimer = null;

        this.reconnectAttempts = 0;

        this.maxReconnectAttempts = Infinity;
    }

    async connect() {

        if (this.stopped) {
            return false;
        }

        if (this.connected) {
            return true;
        }

        if (this.connecting) {
            return false;
        }

        this.connecting = true;

        try {

            const host =
                config.minecraft.host;

            const port =
                config.minecraft.port;

            if (!host) {

                throw new Error(
                    'MC_HOST config.js yoki .env ichida berilmagan.'
                );
            }

            logger.info(
                `Minecraft serverga ulanmoqda: ${host}:${port}`
            );

            this.client =
                bedrock.createClient({

                    host,

                    port,

                    username:
                        config.minecraft.username,

                    offline:
                        config.minecraft.offline,

                    version:
                        config.minecraft.version
                });

            this.registerEvents();

            await this.waitForConnection();

            this.connected = true;

            this.connecting = false;

            this.reconnectAttempts = 0;

            context.state.connected = true;

            context.state.reconnecting = false;

            context.eventBus.emitSafe(
                'minecraft:connected',
                {
                    host,
                    port
                }
            );

            logger.success(
                'Minecraft serverga ulanish muvaffaqiyatli.'
            );

            return true;

        } catch (error) {

            this.connecting = false;

            this.connected = false;

            context.state.connected = false;

            logger.error(
                'Minecraft ulanish xatosi.',
                {
                    error:
                        error.message
                }
            );

            context.eventBus.emitSafe(
                'minecraft:connection_error',
                {
                    error:
                        error.message
                }
            );

            this.scheduleReconnect();

            return false;
        }
    }

    registerEvents() {

        if (!this.client) {
            return;
        }

        this.client.on(
            'spawn',
            () => {

                this.connected = true;

                context.state.connected = true;

                context.state.spawned = true;

                context.state.reconnecting = false;

                logger.success(
                    'Bot Minecraft dunyosida spawn bo‘ldi.'
                );

                context.eventBus.emitSafe(
                    'minecraft:spawn',
                    {}
                );
            }
        );

        this.client.on(
            'text',
            packet => {

                this.handleChat(
                    packet
                );
            }
        );

        this.client.on(
            'player_list',
            packet => {

                context.eventBus.emitSafe(
                    'minecraft:player_list',
                    packet
                );
            }
        );

        this.client.on(
            'add_player',
            packet => {

                context.eventBus.emitSafe(
                    'minecraft:player_join',
                    packet
                );
            }
        );

        this.client.on(
            'remove_player',
            packet => {

                context.eventBus.emitSafe(
                    'minecraft:player_leave',
                    packet
                );
            }
        );

        this.client.on(
            'start_game',
            packet => {

                context.eventBus.emitSafe(
                    'minecraft:start_game',
                    packet
                );
            }
        );

        this.client.on(
            'level_chunk',
            packet => {

                context.eventBus.emitSafe(
                    'minecraft:level_chunk',
                    packet
                );
            }
        );

        this.client.on(
            'update_block',
            packet => {

                context.eventBus.emitSafe(
                    'minecraft:block_update',
                    packet
                );
            }
        );

        this.client.on(
            'move_player',
            packet => {

                context.eventBus.emitSafe(
                    'minecraft:move_player',
                    packet
                );
            }
        );

        this.client.on(
            'add_entity',
            packet => {

                context.eventBus.emitSafe(
                    'minecraft:entity_spawn',
                    packet
                );
            }
        );

        this.client.on(
            'remove_entity',
            packet => {

                context.eventBus.emitSafe(
                    'minecraft:entity_remove',
                    packet
                );
            }
        );

        this.client.on(
            'mob_equipment',
            packet => {

                context.eventBus.emitSafe(
                    'minecraft:mob_equipment',
                    packet
                );
            }
        );

        this.client.on(
            'inventory_content',
            packet => {

                context.eventBus.emitSafe(
                    'minecraft:inventory',
                    packet
                );
            }
        );

        this.client.on(
            'container_open',
            packet => {

                context.eventBus.emitSafe(
                    'minecraft:container_open',
                    packet
                );
            }
        );

        this.client.on(
            'container_close',
            packet => {

                context.eventBus.emitSafe(
                    'minecraft:container_close',
                    packet
                );
            }
        );

        this.client.on(
            'disconnect',
            packet => {

                this.handleDisconnect(
                    packet
                );
            }
        );

        this.client.on(
            'close',
            () => {

                this.handleDisconnect(
                    {
                        reason:
                            'connection closed'
                    }
                );
            }
        );

        this.client.on(
            'error',
            error => {

                logger.error(
                    'Minecraft client error.',
                    {
                        error:
                            error.message
                    }
                );

                context.state.statistics.errors++;

                context.eventBus.emitSafe(
                    'minecraft:error',
                    {
                        error:
                            error.message
                    }
                );
            }
        );
    }

    handleChat(packet) {

        let message = '';

        if (
            packet &&
            typeof packet.message ===
                'string'
        ) {

            message =
                packet.message;
        }

        const sourceName =
            packet?.source_name ||
            packet?.sourceName ||
            packet?.name ||
            '';

        const type =
            packet?.type ||
            'unknown';

        const chat = {

            type,

            sender:
                sourceName,

            message,

            timestamp:
                Date.now(),

            raw:
                packet
        };

        context.state.setLastChat(
            chat
        );

        context.eventBus.emitSafe(
            'minecraft:chat',
            chat
        );
    }

    handleDisconnect(packet) {

        if (!this.connected &&
            !this.connecting) {
            return;
        }

        this.connected = false;

        this.connecting = false;

        context.state.connected = false;

        context.state.spawned = false;

        context.state.reconnecting = true;

        context.state.statistics.reconnects++;

        logger.warn(
            'Minecraft server bilan aloqa uzildi.'
        );

        context.eventBus.emitSafe(
            'minecraft:disconnected',
            {
                packet
            }
        );

        this.scheduleReconnect();
    }

    scheduleReconnect() {

        if (this.stopped) {
            return;
        }

        if (!config.minecraft.autoReconnect) {
            return;
        }

        if (this.reconnectTimer) {
            return;
        }

        this.reconnectAttempts++;

        const delay =
            Math.min(
                config.minecraft.reconnectDelay *
                Math.max(
                    1,
                    this.reconnectAttempts
                ),
                120000
            );

        logger.info(
            `Reconnect ${Math.round(delay / 1000)} soniyadan keyin.`
        );

        this.reconnectTimer =
            setTimeout(
                async () => {

                    this.reconnectTimer = null;

                    await this.connect();

                },
                delay
            );
    }

    waitForConnection() {

        return new Promise(
            (resolve, reject) => {

                if (!this.client) {

                    reject(
                        new Error(
                            'Minecraft client yaratilmagan.'
                        )
                    );

                    return;
                }

                let finished = false;

                const success = () => {

                    if (finished) {
                        return;
                    }

                    finished = true;

                    cleanup();

                    resolve(true);
                };

                const failure = error => {

                    if (finished) {
                        return;
                    }

                    finished = true;

                    cleanup();

                    reject(
                        error instanceof Error
                            ? error
                            : new Error(
                                String(error)
                            )
                    );
                };

                const cleanup = () => {

                    try {

                        this.client.off(
                            'spawn',
                            success
                        );

                    } catch (_) {}

                    try {

                        this.client.off(
                            'error',
                            failure
                        );

                    } catch (_) {}

                    try {

                        this.client.off(
                            'disconnect',
                            failure
                        );

                    } catch (_) {}
                };

                this.client.once(
                    'spawn',
                    success
                );

                this.client.once(
                    'error',
                    failure
                );

                this.client.once(
                    'disconnect',
                    failure
                );

                setTimeout(
                    () => {

                        if (!finished) {

                            failure(
                                new Error(
                                    'Minecraft connection timeout.'
                                )
                            );
                        }

                    },
                    60000
                );
            }
        );
    }

    sendChat(message) {

        if (!this.client) {

            return false;
        }

        if (!this.connected) {

            return false;
        }

        if (
            typeof message !== 'string' ||
            !message.trim()
        ) {

            return false;
        }

        const text =
            message
                .trim()
                .slice(0, 512);

        try {

            this.client.queue(
                'text',
                {
                    type: 'chat',
                    needs_translation: false,
                    source_name:
                        config.minecraft.username,
                    message: text
                }
            );

            context.eventBus.emitSafe(
                'minecraft:chat_sent',
                {
                    message: text
                }
            );

            return true;

        } catch (error) {

            logger.error(
                'Minecraft chat yuborish xatosi.',
                {
                    error:
                        error.message
                }
            );

            return false;
        }
    }

    sendCommand(command) {

        if (!this.client) {
            return false;
        }

        if (!this.connected) {
            return false;
        }

        if (
            typeof command !== 'string' ||
            !command.trim()
        ) {

            return false;
        }

        try {

            this.client.queue(
                'command_request',
                {
                    command:
                        command.trim(),

                    origin: {
                        type: 'player'
                    },

                    internal: false,

                    version: 1
                }
            );

            context.eventBus.emitSafe(
                'minecraft:command_sent',
                {
                    command:
                        command.trim()
                }
            );

            return true;

        } catch (error) {

            logger.error(
                'Minecraft command yuborish xatosi.',
                {
                    error:
                        error.message
                }
            );

            return false;
        }
    }

    getClient() {

        return this.client;
    }

    isConnected() {

        return (
            this.connected &&
            !!this.client
        );
    }

    async disconnect() {

        this.stopped = true;

        if (this.reconnectTimer) {

            clearTimeout(
                this.reconnectTimer
            );

            this.reconnectTimer = null;
        }

        try {

            if (this.client) {

                this.client.close();
            }

        } catch (_) {}

        this.connected = false;

        this.connecting = false;

        context.state.connected = false;

        context.state.spawned = false;

        context.state.reconnecting = false;

        logger.info(
            'Minecraft client to‘xtatildi.'
        );
    }

    status() {

        return {

            connected:
                this.connected,

            connecting:
                this.connecting,

            reconnecting:
                context.state.reconnecting,

            reconnectAttempts:
                this.reconnectAttempts,

            host:
                config.minecraft.host,

            port:
                config.minecraft.port,

            version:
                config.minecraft.version,

            username:
                config.minecraft.username
        };
    }
}

const minecraftClient =
    new MinecraftClient();

module.exports =
    minecraftClient;

module.exports.MinecraftClient =
    MinecraftClient;
