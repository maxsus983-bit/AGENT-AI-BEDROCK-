'use strict';

let TelegramBot = null;

const fallbackLogger = {

    info: console.log,

    warn: console.warn,

    error: console.error,

    success: console.log,

    debug: console.log
};

class TelegramGateway {

    constructor(options = {}) {

        this.name =
            'telegram-gateway';

        this.running =
            false;

        this.bot =
            null;

        this.context =
            null;

        this.token =
            options.token ||
            '';

        this.ownerId =
            options.ownerId ||
            '';

        this.allowedUsers =
            new Set();

        this.chatIds =
            new Set();

        this.startedAt =
            null;
    }

    /* ========================================================
       INITIALIZE
    ======================================================== */

    async initialize(context) {

        this.context =
            context;

        const config =
            context?.config?.telegram || {};

        this.token =
            config.token ||
            process.env.TELEGRAM_BOT_TOKEN ||
            this.token ||
            '';

        this.ownerId =
            String(
                config.ownerId ||
                process.env.TELEGRAM_OWNER_ID ||
                this.ownerId ||
                ''
            ).trim();

        const configuredUsers =
            config.allowedUsers || [];

        this.allowedUsers =
            new Set(
                configuredUsers
                    .map(String)
                    .map(v => v.trim())
                    .filter(Boolean)
            );

        /*
         * Owner ID ham avtomatik ruxsat oladi.
         */

        if (this.ownerId) {

            this.allowedUsers.add(
                this.ownerId
            );
        }

        return this;
    }

    /* ========================================================
       LOGGER
    ======================================================== */

    get logger() {

        return (
            this.context?.logger ||
            fallbackLogger
        );
    }

    /* ========================================================
       START
    ======================================================== */

    async start(context = null) {

        if (context) {

            await this.initialize(
                context
            );
        }

        if (this.running) {

            return {
                success:
                    true,

                alreadyRunning:
                    true
            };
        }

        if (!this.token) {

            this.logger.warn(
                'Telegram token berilmagan.'
            );

            return {

                success:
                    false,

                disabled:
                    true,

                error:
                    'TELEGRAM_BOT_TOKEN topilmadi.'
            };
        }

        try {

            TelegramBot =
                require(
                    'node-telegram-bot-api'
                );

        } catch (error) {

            this.logger.error(
                'node-telegram-bot-api o‘rnatilmagan.'
            );

            this.logger.error(
                'O‘rnating: npm install node-telegram-bot-api'
            );

            return {

                success:
                    false,

                error:
                    'node-telegram-bot-api o‘rnatilmagan.'
            };
        }

        try {

            this.bot =
                new TelegramBot(
                    this.token,
                    {
                        polling: true
                    }
                );

            this.registerHandlers();

            this.running =
                true;

            this.startedAt =
                Date.now();

            let botInfo = null;

            try {

                botInfo =
                    await this.bot.getMe();

            } catch (error) {

                this.logger.warn(
                    `Telegram getMe xatosi: ${error.message}`
                );
            }

            if (botInfo) {

                this.logger.success(
                    `Telegram bot ulandi: @${botInfo.username || botInfo.first_name}`
                );
            } else {

                this.logger.success(
                    'Telegram Gateway ishga tushdi.'
                );
            }

            return {
                success:
                    true,

                bot:
                    botInfo
            };

        } catch (error) {

            this.running =
                false;

            this.bot =
                null;

            this.logger.error(
                `Telegram startup error: ${error.message}`
            );

            return {

                success:
                    false,

                error:
                    error.message
            };
        }
    }

    /* ========================================================
       STOP
    ======================================================== */

    async stop() {

        if (
            this.bot &&
            typeof this.bot.stopPolling ===
            'function'
        ) {

            try {

                await this.bot.stopPolling();

            } catch (error) {

                this.logger.warn(
                    `Telegram polling stop xatosi: ${error.message}`
                );
            }
        }

        this.running =
            false;

        this.bot =
            null;

        return {
            success:
                true
        };
    }

    /* ========================================================
       HANDLERS
    ======================================================== */

    registerHandlers() {

        if (!this.bot) {
            return;
        }

        /*
         * Oddiy xabar.
         */

        this.bot.on(
            'message',
            async message => {

                try {

                    await this.handleMessage(
                        message
                    );

                } catch (error) {

                    this.logger.error(
                        `Telegram message error: ${error.message}`
                    );

                    if (
                        message?.chat?.id
                    ) {

                        await this.safeSend(
                            message.chat.id,
                            `❌ Xatolik: ${error.message}`
                        );
                    }
                }
            }
        );

        /*
         * /start
         */

        this.bot.onText(
            /^\/start(?:\s+(.+))?$/i,
            async message => {

                await this.safeSend(
                    message.chat.id,
                    [
                        '🤖 AKV Minecraft AI Agent',
                        '',
                        '✅ Telegram ulanishi ishlayapti.',
                        '',
                        'Minecraft botni Telegram orqali boshqarishingiz mumkin.',
                        '',
                        'Misollar:',
                        '• oldinga 10 blok yur',
                        '• atrofni kuzat',
                        '• playerlarni ko‘r',
                        '• uy qur',
                        '• jang qil',
                        '• chatga Salom yoz',
                        '• avto o‘yna',
                        '• to‘xta',
                        '',
                        '/status — Agent holati',
                        '/help — Buyruqlar',
                        '/cancel ID — Buyruqni bekor qilish'
                    ].join('\n')
                );
            }
        );

        /*
         * /status
         */

        this.bot.onText(
            /^\/status$/i,
            async message => {

                if (
                    !this.isAllowedUser(
                        message.from
                    )
                ) {

                    await this.safeSend(
                        message.chat.id,
                        '⛔ Sizga ruxsat berilmagan.'
                    );

                    return;
                }

                const status =
                    this.getAgentStatus();

                await this.safeSend(
                    message.chat.id,
                    this.formatStatus(
                        status
                    )
                );
            }
        );

        /*
         * /help
         */

        this.bot.onText(
            /^\/help$/i,
            async message => {

                await this.safeSend(
                    message.chat.id,
                    [
                        '🤖 AKV AI Agent buyruqlari',
                        '',
                        '/start — Botni boshlash',
                        '/status — Agent holati',
                        '/help — Yordam',
                        '/cancel ID — Buyruqni bekor qilish',
                        '',
                        'Oddiy tilda ham buyruq yuborishingiz mumkin:',
                        '',
                        '➡️ oldinga 10 blok yur',
                        '➡️ atrofni kuzat',
                        '➡️ playerlarni ko‘r',
                        '➡️ uy qur',
                        '➡️ jang qil',
                        '➡️ chatga Salom yoz',
                        '➡️ avto o‘yna',
                        '➡️ to‘xta'
                    ].join('\n')
                );
            }
        );

        /*
         * /cancel
         */

        this.bot.onText(
            /^\/cancel(?:\s+(.+))?$/i,
            async message => {

                if (
                    !this.isAllowedUser(
                        message.from
                    )
                ) {

                    await this.safeSend(
                        message.chat.id,
                        '⛔ Sizga ruxsat berilmagan.'
                    );

                    return;
                }

                const match =
                    message.text?.match(
                        /^\/cancel(?:\s+(.+))?$/i
                    );

                const target =
                    match?.[1]?.trim();

                if (!target) {

                    await this.safeSend(
                        message.chat.id,
                        '❌ Masalan: /cancel abc123'
                    );

                    return;
                }

                const executor =
                    this.getModule(
                        'command-executor'
                    );

                const router =
                    this.getModule(
                        'commands'
                    ) ||
                    this.getModule(
                        'command-router'
                    );

                let result = null;

                if (
                    executor &&
                    typeof executor.cancel ===
                    'function'
                ) {

                    result =
                        await executor.cancel(
                            target
                        );

                } else if (
                    router &&
                    typeof router.cancel ===
                    'function'
                ) {

                    result =
                        await router.cancel(
                            target
                        );
                }

                if (!result) {

                    await this.safeSend(
                        message.chat.id,
                        '⚠️ Cancel funksiyasi mavjud emas.'
                    );

                    return;
                }

                await this.safeSend(
                    message.chat.id,
                    result.success
                        ? `✅ Buyruq bekor qilindi: ${target}`
                        : `❌ ${result.error || 'Buyruq topilmadi.'}`
                );
            }
        );
    }

    /* ========================================================
       HANDLE MESSAGE
    ======================================================== */

    async handleMessage(message) {

        if (
            !message ||
            !message.chat
        ) {
            return;
        }

        const chatId =
            String(
                message.chat.id
            );

        this.chatIds.add(
            chatId
        );

        /*
         * Botning o‘z xabarini qayta ishlamaymiz.
         */

        if (
            message.from &&
            message.from.is_bot
        ) {
            return;
        }

        const text =
            message.text ||
            message.caption ||
            '';

        if (!text.trim()) {
            return;
        }

        /*
         * Slash commandlar alohida handler orqali ishlaydi.
         */

        if (
            text.trim().startsWith('/')
        ) {
            return;
        }

        /*
         * Permission.
         */

        if (
            !this.isAllowedUser(
                message.from
            )
        ) {

            await this.safeSend(
                chatId,
                [
                    '⛔ Ruxsat berilmagan.',
                    '',
                    `Telegram User ID: ${message.from?.id || 'unknown'}`
                ].join('\n')
            );

            return;
        }

        /*
         * Buyruqni Agentga yuboramiz.
         */

        const result =
            await this.sendToAgent(
                {
                    text:
                        text.trim(),

                    source:
                        'telegram',

                    channel:
                        'telegram',

                    user:
                        {
                            id:
                                message.from?.id,

                            username:
                                message.from?.username,

                            firstName:
                                message.from?.first_name,

                            lastName:
                                message.from?.last_name,

                            name:
                                [
                                    message.from?.first_name,
                                    message.from?.last_name
                                ]
                                    .filter(Boolean)
                                    .join(' ')
                        },

                    metadata:
                        {
                            chatId,

                            messageId:
                                message.message_id
                        }
                }
            );

        /*
         * Natijani Telegramga qaytaramiz.
         */

        if (!result) {

            await this.safeSend(
                chatId,
                '❌ Agentdan javob olinmadi.'
            );

            return;
        }

        if (result.success) {

            const response =
                this.formatAgentResult(
                    result
                );

            await this.safeSend(
                chatId,
                response
            );

        } else {

            await this.safeSend(
                chatId,
                `❌ ${result.error || 'Buyruq bajarilmadi.'}`
            );
        }
    }

    /* ========================================================
       SEND TO AGENT
    ======================================================== */

    async sendToAgent(request) {

        /*
         * 1. Command Executor
         */

        const executor =
            this.getModule(
                'commandExecutor'
            ) ||
            this.getModule(
                'command-executor'
            );

        if (
            executor &&
            typeof executor.handleExternal ===
            'function'
        ) {

            return await executor.handleExternal(
                {
                    command:
                        request.text,

                    source:
                        'telegram',

                    channel:
                        'telegram',

                    user:
                        request.user,

                    metadata:
                        request.metadata,

                    requestId:
                        null
                }
            );
        }

        /*
         * 2. Commands / Router
         */

        const router =
            this.getModule(
                'commands'
            ) ||
            this.getModule(
                'command-router'
            );

        if (
            router &&
            typeof router.handle ===
            'function'
        ) {

            return await router.handle(
                request.text,
                'telegram',
                request.user
            );
        }

        /*
         * 3. Generic receive()
         */

        const commandGateway =
            this.getModule(
                'command-gateway'
            );

        if (
            commandGateway &&
            typeof commandGateway.receive ===
            'function'
        ) {

            return await commandGateway.receive(
                request
            );
        }

        return {

            success:
                false,

            error:
                'Command Executor yoki Command Router topilmadi.'
        };
    }

    /* ========================================================
       MODULE GETTER
    ======================================================== */

    getModule(name) {

        if (
            this.context &&
            typeof this.context.get ===
            'function'
        ) {

            const result =
                this.context.get(
                    name
                );

            if (result) {
                return result;
            }
        }

        if (
            this.context?.modules?.[name]
        ) {

            return this.context.modules[name];
        }

        if (
            this.context?.registry?.[name]
        ) {

            return this.context.registry[name];
        }

        return null;
    }

    /* ========================================================
       PERMISSION
    ======================================================== */

    isAllowedUser(user) {

        /*
         * Hech qanday restriction berilmagan bo‘lsa,
         * Telegramdagi foydalanuvchini qabul qilamiz.
         */

        if (
            this.allowedUsers.size === 0
        ) {
            return true;
        }

        const values = [

            user?.id,

            user?.username,

            user?.first_name,

            user?.last_name

        ]
            .filter(
                value =>
                    value !== undefined &&
                    value !== null
            )
            .map(
                String
            );

        return values.some(
            value =>
                this.allowedUsers.has(
                    value
                )
        );
    }

    /* ========================================================
       AGENT STATUS
    ======================================================== */

    getAgentStatus() {

        const state =
            this.context?.state || {};

        const executor =
            this.getModule(
                'commandExecutor'
            ) ||
            this.getModule(
                'command-executor'
            );

        let executorStatus =
            null;

        try {

            if (
                executor &&
                typeof executor.status ===
                'function'
            ) {

                executorStatus =
                    executor.status();
            }

        } catch (_) {}

        return {

            running:
                true,

            telegram:
                this.running,

            minecraft:
                state.connected === true,

            spawned:
                state.spawned === true,

            autonomous:
                state.autonomousMode === true,

            currentTask:
                state.currentTask,

            currentGoal:
                state.currentGoal,

            position:
                state.position,

            queue:
                state.taskQueue?.length || 0,

            executor:
                executorStatus,

            statistics:
                state.statistics || {}
        };
    }

    /* ========================================================
       FORMAT STATUS
    ======================================================== */

    formatStatus(status) {

        const stats =
            status.statistics || {};

        const position =
            status.position
                ? `${status.position.x ?? '?'}, ${status.position.y ?? '?'}, ${status.position.z ?? '?'}`
                : 'noma’lum';

        return [

            '🤖 AKV AI AGENT',

            '',

            `📡 Telegram: ${status.telegram ? '🟢' : '🔴'}`,

            `⛏ Minecraft: ${status.minecraft ? '🟢' : '🔴'}`,

            `🌍 Spawned: ${status.spawned ? '🟢' : '🔴'}`,

            `🧠 Autonomous: ${status.autonomous ? '🟢' : '🔴'}`,

            `📍 Position: ${position}`,

            `📋 Queue: ${status.queue}`,

            `🎯 Task: ${status.currentTask || 'yo‘q'}`,

            `🎯 Goal: ${status.currentGoal || 'yo‘q'}`,

            '',

            '📊 STATISTICS',

            `💬 Messages: ${stats.messagesSeen || 0}`,

            `👤 Players: ${stats.playersSeen || 0}`,

            `👾 Entities: ${stats.entitiesSeen || 0}`,

            `🧠 Decisions: ${stats.decisionsMade || 0}`,

            `✅ Completed: ${stats.tasksCompleted || 0}`,

            `❌ Failed: ${stats.tasksFailed || 0}`,

            `⚔️ Attacks: ${stats.attacks || 0}`,

            `🔄 Reconnects: ${stats.reconnects || 0}`

        ].join('\n');
    }

    /* ========================================================
       FORMAT RESULT
    ======================================================== */

    formatAgentResult(result) {

        const lines = [];

        if (
            result.message
        ) {

            lines.push(
                String(
                    result.message
                )
            );
        }

        if (
            result.response
        ) {

            lines.push(
                String(
                    result.response
                )
            );
        }

        if (
            result.requestId
        ) {

            lines.push(
                `🆔 ${result.requestId}`
            );
        }

        if (
            result.command
        ) {

            const type =
                result.command.type ||
                result.command.name ||
                'AI';

            lines.push(
                `⚙️ ${type}`
            );
        }

        if (
            lines.length === 0
        ) {

            lines.push(
                '✅ Buyruq qabul qilindi.'
            );
        }

        return lines.join('\n');
    }

    /* ========================================================
       SEND
    ======================================================== */

    async send(message) {

        if (!this.bot) {

            return {

                success:
                    false,

                error:
                    'Telegram bot ishlamayapti.'
            };
        }

        const targets =
            message.chatId
                ? [String(message.chatId)]
                : [...this.chatIds];

        if (
            targets.length === 0
        ) {

            return {

                success:
                    false,

                error:
                    'Telegram chat ID topilmadi.'
            };
        }

        const results = [];

        for (
            const chatId of targets
        ) {

            results.push(
                await this.safeSend(
                    chatId,
                    String(
                        message.text ||
                        message.message ||
                        ''
                    )
                )
            );
        }

        return {

            success:
                true,

            results
        };
    }

    /* ========================================================
       NOTIFY
    ======================================================== */

    async notify(text) {

        return this.send({
            text
        });
    }

    /* ========================================================
       SAFE SEND
    ======================================================== */

    async safeSend(chatId, text) {

        if (!this.bot) {

            return {
                success:
                    false
            };
        }

        try {

            const chunks =
                this.splitMessage(
                    String(text || '')
                );

            const results = [];

            for (
                const chunk of chunks
            ) {

                const result =
                    await this.bot.sendMessage(
                        chatId,
                        chunk,
                        {
                            disable_web_page_preview:
                                true
                        }
                    );

                results.push(
                    result
                );
            }

            return {

                success:
                    true,

                results
            };

        } catch (error) {

            this.logger.warn(
                `Telegram xabar yuborilmadi: ${error.message}`
            );

            return {

                success:
                    false,

                error:
                    error.message
            };
        }
    }

    /* ========================================================
       SPLIT MESSAGE
    ======================================================== */

    splitMessage(text) {

        const limit =
            3900;

        if (
            text.length <= limit
        ) {

            return [text];
        }

        const chunks = [];

        for (
            let i = 0;
            i < text.length;
            i += limit
        ) {

            chunks.push(
                text.slice(
                    i,
                    i + limit
                )
            );
        }

        return chunks;
    }

    /* ========================================================
       STATUS
    ======================================================== */

    getStatus() {

        return {

            running:
                this.running,

            connected:
                Boolean(this.bot),

            chats:
                this.chatIds.size,

            allowedUsers:
                this.allowedUsers.size,

            startedAt:
                this.startedAt
        };
    }
}

/* ============================================================
   SINGLETON
============================================================ */

const telegramGateway =
    new TelegramGateway();

module.exports =
    telegramGateway;

module.exports.TelegramGateway =
    TelegramGateway;
