'use strict';

const https = require('https');
const logger = require('../core/logger');
const context = require('../core/agent-context');

class OpenRouter {

    constructor() {

        this.apiKey =
            process.env.OPENROUTER_API_KEY ||
            '';

        this.baseURL =
            'openrouter.ai';

        this.endpoint =
            '/api/v1/chat/completions';

        this.model =
            process.env.OPENROUTER_MODEL ||
            'openrouter/free';

        this.fallbackModels = [

            'openrouter/free',

            'google/gemini-2.5-flash',

            'meta-llama/llama-3.3-70b-instruct',

            'openai/gpt-4o-mini'
        ];

        this.timeout =
            Number(
                process.env.OPENROUTER_TIMEOUT ||
                60000
            );

        this.maxRetries =
            Number(
                process.env.OPENROUTER_RETRIES ||
                2
            );

        this.siteURL =
            process.env.OPENROUTER_SITE_URL ||
            'http://localhost';

        this.siteName =
            process.env.OPENROUTER_SITE_NAME ||
            'Minecraft AI Agent';

        this.totalRequests = 0;
        this.successfulRequests = 0;
        this.failedRequests = 0;

        this.lastModel = null;
        this.lastError = null;
        this.lastRequestAt = null;
        this.lastResponseAt = null;

        context.register(
            'openrouter',
            this
        );
    }

    configure(options = {}) {

        if (
            options.apiKey
        ) {

            this.apiKey =
                String(
                    options.apiKey
                ).trim();
        }

        if (
            options.model
        ) {

            this.model =
                String(
                    options.model
                ).trim();
        }

        if (
            Array.isArray(
                options.fallbackModels
            ) &&
            options.fallbackModels.length
        ) {

            this.fallbackModels =
                options.fallbackModels
                    .map(
                        model =>
                            String(
                                model
                            ).trim()
                    )
                    .filter(
                        Boolean
                    );
        }

        return this.status();
    }

    isConfigured() {

        return Boolean(
            this.apiKey
        );
    }

    async complete(options = {}) {

        const system =
            options.system ||
            '';

        const prompt =
            options.prompt ||
            '';

        const temperature =
            Number.isFinite(
                Number(
                    options.temperature
                )
            )
                ? Number(
                    options.temperature
                )
                : 0.2;

        const maxTokens =
            Number.isFinite(
                Number(
                    options.maxTokens
                )
            )
                ? Number(
                    options.maxTokens
                )
                : 1800;

        return this.chat({

            messages: [

                {
                    role:
                        'system',

                    content:
                        system
                },

                {
                    role:
                        'user',

                    content:
                        prompt
                }
            ],

            temperature,

            max_tokens:
                maxTokens
        });
    }

    async chat(options = {}) {

        if (
            !this.isConfigured()
        ) {

            throw new Error(
                'OPENROUTER_API_KEY sozlanmagan.'
            );
        }

        const messages =
            Array.isArray(
                options.messages
            )
                ? options.messages
                : [];

        if (
            !messages.length
        ) {

            throw new Error(
                'OpenRouter uchun messages bo‘sh.'
            );
        }

        const requestedModel =
            options.model ||
            this.model;

        const models =
            this.buildModelList(
                requestedModel
            );

        let lastError =
            null;

        for (
            const model of models
        ) {

            for (
                let attempt = 0;
                attempt <= this.maxRetries;
                attempt++
            ) {

                try {

                    this.totalRequests++;

                    this.lastRequestAt =
                        Date.now();

                    const result =
                        await this.request(
                            {
                                model,

                                messages,

                                temperature:
                                    options.temperature ??
                                    0.2,

                                max_tokens:
                                    options.max_tokens ??
                                    1800,

                                top_p:
                                    options.top_p ??
                                    1,

                                stream:
                                    false
                            }
                        );

                    this.successfulRequests++;

                    this.lastModel =
                        model;

                    this.lastResponseAt =
                        Date.now();

                    this.lastError =
                        null;

                    return result;

                } catch (error) {

                    lastError =
                        error;

                    this.lastError =
                        error.message;

                    logger.warn(
                        `OpenRouter model ${model} xato: ${error.message}`
                    );

                    if (
                        !this.shouldRetry(
                            error
                        )
                    ) {

                        break;
                    }

                    if (
                        attempt <
                        this.maxRetries
                    ) {

                        await this.sleep(
                            this.retryDelay(
                                attempt
                            )
                        );
                    }
                }
            }
        }

        this.failedRequests++;

        throw (
            lastError ||
            new Error(
                'OpenRouter so‘rovi bajarilmadi.'
            )
        );
    }

    buildModelList(
        requestedModel
    ) {

        const list = [];

        if (
            requestedModel
        ) {

            list.push(
                requestedModel
            );
        }

        for (
            const model of
            this.fallbackModels
        ) {

            if (
                model &&
                !list.includes(
                    model
                )
            ) {

                list.push(
                    model
                );
            }
        }

        return list;
    }

    request(
        payload
    ) {

        return new Promise(
            (
                resolve,
                reject
            ) => {

                const body =
                    JSON.stringify(
                        payload
                    );

                const request =
                    https.request(
                        {
                            hostname:
                                this.baseURL,

                            path:
                                this.endpoint,

                            method:
                                'POST',

                            timeout:
                                this.timeout,

                            headers: {

                                'Authorization':
                                    `Bearer ${this.apiKey}`,

                                'Content-Type':
                                    'application/json',

                                'Content-Length':
                                    Buffer.byteLength(
                                        body
                                    ),

                                'HTTP-Referer':
                                    this.siteURL,

                                'X-Title':
                                    this.siteName
                            }
                        },

                        response => {

                            let data =
                                '';

                            response.setEncoding(
                                'utf8'
                            );

                            response.on(
                                'data',
                                chunk => {

                                    data +=
                                        chunk;
                                }
                            );

                            response.on(
                                'end',
                                () => {

                                    let parsed =
                                        null;

                                    try {

                                        parsed =
                                            JSON.parse(
                                                data
                                            );

                                    } catch (error) {

                                        reject(
                                            new Error(
                                                `OpenRouter JSON xatosi. HTTP ${response.statusCode}`
                                            )
                                        );

                                        return;
                                    }

                                    if (
                                        response.statusCode >=
                                        200 &&
                                        response.statusCode <
                                        300
                                    ) {

                                        resolve(
                                            parsed
                                        );

                                        return;
                                    }

                                    const message =
                                        parsed
                                            ?.error
                                            ?.message ||
                                        parsed
                                            ?.message ||
                                        `HTTP ${response.statusCode}`;

                                    const error =
                                        new Error(
                                            message
                                        );

                                    error.statusCode =
                                        response.statusCode;

                                    error.response =
                                        parsed;

                                    reject(
                                        error
                                    );
                                }
                            );
                        }
                    );

                request.on(
                    'timeout',
                    () => {

                        request.destroy();

                        const error =
                            new Error(
                                'OpenRouter timeout.'
                            );

                        error.code =
                            'ETIMEDOUT';

                        reject(
                            error
                        );
                    }
                );

                request.on(
                    'error',
                    error => {

                        reject(
                            error
                        );
                    }
                );

                request.write(
                    body
                );

                request.end();
            }
        );
    }

    shouldRetry(
        error
    ) {

        if (
            !error
        ) {

            return false;
        }

        if (
            error.code ===
            'ETIMEDOUT'
        ) {

            return true;
        }

        if (
            error.code ===
            'ECONNRESET'
        ) {

            return true;
        }

        const status =
            Number(
                error.statusCode
            );

        if (
            status === 429
        ) {

            return true;
        }

        if (
            status >= 500
        ) {

            return true;
        }

        return false;
    }

    retryDelay(
        attempt
    ) {

        const base =
            1000;

        return Math.min(
            15000,
            base *
            Math.pow(
                2,
                attempt
            )
        );
    }

    extractText(
        response
    ) {

        if (
            typeof response ===
            'string'
        ) {

            return response;
        }

        if (
            response &&
            response.choices &&
            response.choices[0]
        ) {

            const choice =
                response.choices[0];

            if (
                choice.message &&
                typeof choice.message.content ===
                'string'
            ) {

                return choice.message.content;
            }

            if (
                typeof choice.text ===
                'string'
            ) {

                return choice.text;
            }
        }

        return '';
    }

    getUsage(
        response
    ) {

        if (
            !response
        ) {

            return null;
        }

        return (
            response.usage ||
            null
        );
    }

    status() {

        return {

            configured:
                this.isConfigured(),

            model:
                this.model,

            fallbackModels:
                this.fallbackModels.slice(),

            totalRequests:
                this.totalRequests,

            successfulRequests:
                this.successfulRequests,

            failedRequests:
                this.failedRequests,

            lastModel:
                this.lastModel,

            lastRequestAt:
                this.lastRequestAt,

            lastResponseAt:
                this.lastResponseAt,

            lastError:
                this.lastError
        };
    }

    async test() {

        const response =
            await this.complete({

                system:
                    'You are a connection test assistant. Answer briefly.',

                prompt:
                    'Reply only with: CONNECTION_OK',

                temperature:
                    0,

                maxTokens:
                    20
            });

        return {

            success: true,

            model:
                this.lastModel,

            text:
                this.extractText(
                    response
                ),

            usage:
                this.getUsage(
                    response
                )
        };
    }

    sleep(
        ms
    ) {

        return new Promise(
            resolve =>
                setTimeout(
                    resolve,
                    ms
                )
        );
    }
}

const openrouter =
    new OpenRouter();

module.exports =
    openrouter;

module.exports.OpenRouter =
    OpenRouter;
